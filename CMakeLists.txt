cmake_minimum_required(VERSION 3.16)
project(multisensor_acquisition_studio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(acquisition INTERFACE)
target_include_directories(acquisition INTERFACE include)
target_link_libraries(acquisition INTERFACE Threads::Threads)
add_executable(acquisition_demo src/demo.cpp)
target_link_libraries(acquisition_demo PRIVATE acquisition)
enable_testing()
add_executable(core_checks tests/core_checks.cpp)
target_link_libraries(core_checks PRIVATE acquisition)
add_test(NAME core_checks COMMAND core_checks)
option(WITH_QT "Build optional Qt 5 synthetic monitor" OFF)
if(WITH_QT)
  find_package(Qt5 REQUIRED COMPONENTS Widgets)
  set(CMAKE_AUTOMOC ON)
  add_executable(status_window adapters/qt/status_window.cpp)
  target_link_libraries(status_window PRIVATE acquisition Qt5::Widgets)
endif()
option(WITH_ROS1 "Build optional ROS 1 adapter in a sourced ROS environment" OFF)
if(WITH_ROS1)
  find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs std_msgs)
  add_executable(snapshot_node adapters/ros1/snapshot_node.cpp)
  target_include_directories(snapshot_node PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(snapshot_node PRIVATE acquisition ${catkin_LIBRARIES})
endif()
